#include <Wire.h>

const uint8_t MPU_ADDR = 0x68;

const int SDA_PIN = 21;
const int SCL_PIN = 22;

float gyroOffsetX = 0;
float gyroOffsetY = 0;
float gyroOffsetZ = 0;

const float GYRO_SCALE = 65.5;

const float START_THRESHOLD = 35.0;
const float STOP_THRESHOLD = 12.0;

int repCount = 0;

bool movingUp = false;
bool waitingForReturn = false;

unsigned long lastRepTime = 0;
const unsigned long REP_LOCKOUT = 500;

void writeMPU(uint8_t reg, uint8_t value) {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(reg);
  Wire.write(value);
  Wire.endTransmission();
}

bool readMPU(
  int16_t &ax,
  int16_t &ay,
  int16_t &az,
  int16_t &gx,
  int16_t &gy,
  int16_t &gz
) {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B);

  if (Wire.endTransmission(false) != 0) {
    return false;
  }

  if (Wire.requestFrom((int)MPU_ADDR, 14) != 14) {
    return false;
  }

  ax = (Wire.read() << 8) | Wire.read();
  ay = (Wire.read() << 8) | Wire.read();
  az = (Wire.read() << 8) | Wire.read();

  Wire.read();
  Wire.read();

  gx = (Wire.read() << 8) | Wire.read();
  gy = (Wire.read() << 8) | Wire.read();
  gz = (Wire.read() << 8) | Wire.read();

  return true;
}

void calibrateGyro() {
  Serial.println("Keep wrist completely still...");
  Serial.println("Calibrating MPU...");

  long sumX = 0;
  long sumY = 0;
  long sumZ = 0;

  int samples = 500;
  int validSamples = 0;

  for (int i = 0; i < samples; i++) {
    int16_t ax, ay, az;
    int16_t gx, gy, gz;

    if (readMPU(ax, ay, az, gx, gy, gz)) {
      sumX += gx;
      sumY += gy;
      sumZ += gz;
      validSamples++;
    }

    delay(5);
  }

  if (validSamples > 0) {
    gyroOffsetX = sumX / (float)validSamples;
    gyroOffsetY = sumY / (float)validSamples;
    gyroOffsetZ = sumZ / (float)validSamples;
  }

  Serial.println("Calibration complete");
  Serial.println();
}

void setup() {
  Serial.begin(115200);
  delay(1500);

  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setClock(400000);

  Wire.beginTransmission(MPU_ADDR);

  if (Wire.endTransmission() != 0) {
    Serial.println("MPU6050 NOT FOUND");
    while (true) {
      delay(1000);
    }
  }

  Serial.println("MPU6050 FOUND");

  writeMPU(0x6B, 0x00);
  delay(100);

  writeMPU(0x1A, 0x03);

  writeMPU(0x1B, 0x08);

  writeMPU(0x1C, 0x08);

  calibrateGyro();

  Serial.println("Start wrist exercise");
  Serial.println();
  Serial.println("GX,GY,GZ,STATE,REP");
}

void loop() {
  int16_t ax, ay, az;
  int16_t gxRaw, gyRaw, gzRaw;

  if (!readMPU(
        ax,
        ay,
        az,
        gxRaw,
        gyRaw,
        gzRaw
      )) {

    Serial.println("MPU READ ERROR");
    delay(100);
    return;
  }

  float gx =
    (gxRaw - gyroOffsetX) / GYRO_SCALE;

  float gy =
    (gyRaw - gyroOffsetY) / GYRO_SCALE;

  float gz =
    (gzRaw - gyroOffsetZ) / GYRO_SCALE;

  float wristGyro = gy;

  String state = "REST";

  if (!movingUp && !waitingForReturn) {

    if (wristGyro > START_THRESHOLD) {
      movingUp = true;
      state = "WRIST_UP";
    }

  } else if (movingUp) {

    state = "WRIST_UP";

    if (wristGyro < STOP_THRESHOLD) {
      movingUp = false;
      waitingForReturn = true;
    }

  } else if (waitingForReturn) {

    state = "RETURN";

    if (wristGyro < -START_THRESHOLD) {

      if (millis() - lastRepTime > REP_LOCKOUT) {
        repCount++;
        lastRepTime = millis();

        Serial.print("REP COMPLETED: ");
        Serial.println(repCount);
      }

      waitingForReturn = false;
    }
  }

  Serial.print(gx, 1);
  Serial.print(",");

  Serial.print(gy, 1);
  Serial.print(",");

  Serial.print(gz, 1);
  Serial.print(",");

  Serial.print(state);
  Serial.print(",");

  Serial.println(repCount);

  delay(20);
}
